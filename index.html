-- Promedio por programa académico
SELECT  p.Nombre,
        AVG(r.Puntaje) AS Promedio
FROM    Respuestas r
JOIN    Programas  p ON p.Id = r.ProgramaId
GROUP BY p.Nombre
ORDER BY Promedio ASC;
